import io, json, os, sys
sys.path.insert(0, '/workspace/src')
from dotenv import dotenv_values, load_dotenv
case = json.load(sys.stdin)
os.environ.clear()
if case == 'alternate':
    for value, expected in [('present', 'chosen'), ('', ''), (None, '')]:
        if value is not None: os.environ['BASE'] = value
        else: os.environ.pop('BASE', None)
        got = dotenv_values(stream=io.StringIO('OUT=before${BASE:+chosen}after\n'))
        assert got['OUT'] == 'before' + expected + 'after', got
elif case == 'precedence':
    for override, external, local, expected in [(False, '', 'file', ''), (True, '', 'file', 'chosen'),
                                               (False, 'env', '', 'chosen'), (True, 'env', '', '')]:
        os.environ.clear(); os.environ['BASE'] = external
        load_dotenv(stream=io.StringIO('BASE=' + local + '\nOUT=${BASE:+chosen}\n'), override=override)
        assert os.environ['OUT'] == expected, dict(os.environ)
elif case == 'legacy':
    os.environ['BASE'] = ''
    got = dotenv_values(stream=io.StringIO('A=${BASE:-fallback}\nB=${MISSING:-fallback}\nC=${BASE}\nD=$BASE\nE\n'))
    assert got == dict(A='', B='fallback', C='', D='$BASE', E=None), got
elif case == 'literal':
    os.environ.update(BASE='yes', OTHER='expanded')
    got = dotenv_values(stream=io.StringIO('A=${BASE:+$OTHER}\nB=${BASE:+}\nC=x${BASE:+α}${BASE:+β}y\nD=${BASE:+unfinished\n'))
    assert got == dict(A='$OTHER', B='', C='xαβy', D='${BASE:+unfinished'), got
elif case == 'disabled-and-unassigned':
    got = dotenv_values(stream=io.StringIO('BASE\nA=${BASE:+chosen}\nB=${ABSENT:+chosen}\n'))
    assert got == dict(BASE=None, A='', B=''), got
    got = dotenv_values(stream=io.StringIO('A=${BASE:+chosen}\n'), interpolate=False)
    assert got['A'] == '${BASE:+chosen}', got
else: raise AssertionError(case)
print('ok')
