from packaging.specifiers import SpecifierSet
from packaging.version import Version
case = json.load(sys.stdin)
def check(spec, values, expected, constructor=None, call=None):
    original = list(values)
    seen = []
    def once():
        for item in original:
            seen.append(item)
            yield item
    result = list(SpecifierSet(spec, prereleases=constructor).filter(once(), prereleases=call))
    assert [str(v) for v in result] == expected, result
    assert seen == original and all(any(v is item for item in original) for v in result)
if case == 'fallback':
    check('>=1,<2', ['0.9', '1.5a1', '2.1'], ['1.5a1'])

elif case == 'matching-finals':
    check('>=1,<2', ['1.5a1', '1.4', '2.1', '1.6rc1', '1.4'], ['1.4', '1.4'])
    check('>=1,<2', ['1.5a1', '1.4.post1'], ['1.4.post1'])
