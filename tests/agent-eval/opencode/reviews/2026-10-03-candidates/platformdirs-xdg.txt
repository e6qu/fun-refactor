Task: platformdirs-xdg
Requirement:
On Unix, ignore relative XDG_DATA_HOME, XDG_CONFIG_HOME, XDG_CACHE_HOME and XDG_STATE_HOME values and use their existing defaults. For XDG_DATA_DIRS and XDG_CONFIG_DIRS, discard empty and relative entries individually; if no absolute entries remain, use the existing defaults. Preserve absolute paths verbatim, including spaces, input order and duplicates. Keep app/version suffixes, multipath selection, Path wrappers, directory iterators and ensure_exists consistent. Never create a directory from a rejected relative value. Do not change runtime, media or non-Unix directory behavior.

import json, os, sys, types
sys.path.insert(0, '/workspace/src')
sys.modules['platformdirs.version'] = types.SimpleNamespace(__version__='4.3.6', __version_tuple__=(4, 3, 6))

from pathlib import Path
from platformdirs import user_config_dir, site_config_dir, user_config_path
from platformdirs.unix import Unix
case = json.load(sys.stdin)
os.environ.clear()
os.environ['HOME'] = '/tmp/fixture-home'
if case == 'user-homes':
    for kind, fallback in [('data', '.local/share'), ('config', '.config'), ('cache', '.cache'), ('state', '.local/state')]:
        key = 'XDG_' + kind.upper() + '_HOME'
        for value in ['', 'relative', '  relative', './nested']:
            os.environ[key] = value
            assert getattr(Unix('demo', version='2'), 'user_' + kind + '_dir') == '/tmp/fixture-home/' + fallback + '/demo/2'
        os.environ[key] = '/tmp/absolute space'
        assert getattr(Unix('demo', version='2'), 'user_' + kind + '_dir') == '/tmp/absolute space/demo/2'
elif case == 'site-mixed':
    for kind in ['data', 'config']:
        os.environ['XDG_' + kind.upper() + '_DIRS'] = 'relative:/tmp/one::/tmp/two:/tmp/one'
        instance = Unix('demo', version='2', multipath=True)
        value = getattr(instance, 'site_' + kind + '_dir')
        assert value == '/tmp/one/demo/2:/tmp/two/demo/2:/tmp/one/demo/2', value
        assert getattr(instance, 'site_' + kind + '_path') == Path('/tmp/one/demo/2')
        assert getattr(Unix('demo', version='2'), 'site_' + kind + '_dir') == '/tmp/one/demo/2'
elif case == 'site-defaults':
    for value in ['', ':relative::', 'relative:also-relative']:
        os.environ['XDG_DATA_DIRS'] = value; os.environ['XDG_CONFIG_DIRS'] = value
        instance = Unix('demo', version='2', multipath=True)
        assert instance.site_data_dir == '/usr/local/share/demo/2:/usr/share/demo/2'
        assert instance.site_config_dir == '/etc/xdg/demo/2'
elif case == 'wrappers-and-iterators':
    os.environ.update(XDG_CONFIG_HOME='relative', XDG_CONFIG_DIRS='bad:/tmp/valid')
    assert user_config_dir('demo', version='2') == '/tmp/fixture-home/.config/demo/2'
    assert user_config_path('demo', version='2') == Path('/tmp/fixture-home/.config/demo/2')
    assert site_config_dir('demo', version='2', multipath=True) == '/tmp/valid/demo/2'
    assert list(Unix('demo', version='2').iter_config_dirs()) == ['/tmp/fixture-home/.config/demo/2', '/tmp/valid/demo/2']
elif case == 'creation-and-spaces':
    os.chdir('/tmp')
    os.environ['XDG_CONFIG_HOME'] = 'rejected-relative'
    instance = Unix('demo', version='2', ensure_exists=True)
    assert Path(instance.user_config_dir).is_dir()
    assert not Path('/tmp/rejected-relative').exists()
    os.environ['XDG_CONFIG_DIRS'] = '/tmp/space name :/tmp/other'
    assert Unix(multipath=True).site_config_dir == '/tmp/space name :/tmp/other'
elif case == 'absolute-compatibility':
    os.environ.update(XDG_DATA_HOME='/tmp/data', XDG_CONFIG_HOME='/tmp/config', XDG_CACHE_HOME='/tmp/cache',
                      XDG_STATE_HOME='/tmp/state', XDG_DATA_DIRS='/tmp/site1:/tmp/site2', XDG_CONFIG_DIRS='/tmp/configs')
    instance = Unix('demo', version='2', multipath=True)
    assert instance.user_data_dir == '/tmp/data/demo/2'
    assert instance.user_cache_dir == '/tmp/cache/demo/2'
    assert instance.user_log_dir == '/tmp/state/demo/2/log'
    assert list(instance.iter_data_dirs()) == ['/tmp/data/demo/2', '/tmp/site1/demo/2', '/tmp/site2/demo/2']
    assert instance.site_config_dir == '/tmp/configs/demo/2'
else: raise AssertionError(case)
print('ok')
